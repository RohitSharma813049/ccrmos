import { test, expect } from '@playwright/test';

async function login(page: any, email: string) {
  await page.goto('/login');
  await page.fill('input[type="email"]', email);
  await page.click('button:has-text("Send OTP")');
  await expect(page.getByText('Enter OTP')).toBeVisible({ timeout: 15000 });
  await page.fill('input[type="text"]', '123456');
  await page.click('button:has-text("Sign In")');
  await page.waitForURL('**/dashboard**');
}

test.describe('Dynamic Form Builder', () => {
  test('Founder can add a custom field to the Leads module', async ({ page }) => {
    await login(page, 'founderA@crmos.com');
    
    // Navigate to Custom Fields settings
    await page.goto('/dashboard/settings/fields');
    
    // Create new field
    await page.click('button:has-text("Add Field")');
    
    const fieldName = `CustomE2EField_${Date.now()}`;
    await page.fill('input[name="name"]', fieldName);
    await page.selectOption('select[name="type"]', 'Text');
    await page.click('button:has-text("Save Field")');
    
    // Verify field appears in list
    await expect(page.getByText(fieldName)).toBeVisible();
    
    // Go to Leads module to verify it appears in the create form
    await page.click('text=Lead Management');
    await page.click('button:has-text("Add Lead")');
    
    // The custom field label should be visible
    await expect(page.getByText(fieldName)).toBeVisible();
  });
});
